@startuml
skinparam {
  ActorBackgroundColor #FFFFFF
  BoundaryBackgroundColor #FFFFFF
  ControlBackgroundColor #FFFFFF
  EntityBackgroundColor #FFFFFF
  DatabaseBackgroundColor #FFFFFF
  ActorBorderColor black
  BoundaryBorderColor black
  ControlBorderColor black
  EntityBorderColor black
  DatabaseBorderColor black
}

actor "Content Moderator" as Moderator
boundary "Content Management UI" as UI
boundary Router
control MangaController
control MangaService
entity MangaModel
database MongoDB

Moderator -> UI: 1. Click "Approve"
activate Moderator
activate UI

UI -> Router: 2. PATCH /api/manga/admin/license/{mangaId}/review\n{status:"approved", publishAfterApprove:true}
activate Router

Router -> MangaController: 3. reviewLicense(mangaId, dto, req)
activate MangaController

MangaController -> MangaService: 4. reviewLicense(mangaId, reviewerId, approved, "", true)
activate MangaService

opt Invalid mangaId / reviewerId / review status
  MangaService --> MangaController: 4.1. Throw BadRequest
  MangaController --> Router: 4.1.1. HTTP 400
  Router --> UI: 4.1.2. Approve failed
  UI --> Moderator: 4.1.3. Show error message
end

MangaService -> MangaModel: 5. findById(mangaId)
activate MangaModel
MangaModel -> MongoDB: 6. Query story
activate MongoDB
MongoDB --> MangaModel: 7. Manga document / null
deactivate MongoDB
MangaModel --> MangaService: 8. Return manga
deactivate MangaModel

opt Manga not found
  MangaService --> MangaController: 8.1. Throw NotFound
  MangaController --> Router: 8.1.1. HTTP 404
  Router --> UI: 8.1.2. Approve failed
  UI --> Moderator: 8.1.3. Show error message
end

opt License is not pending
  MangaService --> MangaController: 8.2. Throw BadRequest("License is not in pending state")
  MangaController --> Router: 8.2.1. HTTP 400
  Router --> UI: 8.2.2. Approve failed
  UI --> Moderator: 8.2.3. Show error message
end

opt publishAfterApprove = true AND story suspended/banned
  MangaService --> MangaController: 8.3. Throw BadRequest("Cannot publish after approve...")
  MangaController --> Router: 8.3.1. HTTP 400
  Router --> UI: 8.3.2. Approve failed
  UI --> Moderator: 8.3.3. Show error message
end

MangaService -> MangaModel: 9. save(licenseStatus=approved,\nlicenseReviewedAt, licenseReviewedBy,\nlicenseRejectReason="", isPublish=true)
activate MangaModel
MangaModel -> MongoDB: 10. Update document
activate MongoDB
MongoDB --> MangaModel: 11. Saved
deactivate MongoDB
MangaModel --> MangaService: 12. Updated manga
deactivate MangaModel

MangaService --> MangaController: 13. Return success payload
deactivate MangaService

MangaController --> Router: 14. HTTP 200 response
deactivate MangaController

Router --> UI: 15. Approve success
deactivate Router

UI --> Moderator: 16. Show approved result
deactivate UI
deactivate Moderator

@enduml
