@startuml
title Subsequence - Validate ids and find manga

skinparam {
  ActorBackgroundColor #FFFFFF
  BoundaryBackgroundColor #FFFFFF
  ControlBackgroundColor #FFFFFF
  EntityBackgroundColor #FFFFFF
  DatabaseBackgroundColor #FFFFFF
  ActorBorderColor black
  BoundaryBorderColor black
  ControlBorderColor black
  EntityBorderColor black
  DatabaseBorderColor black
}

control MangaService as SV
entity MangaModel as MM
database MongoDB as DB

activate SV

SV -> SV: 1. Validate mangaId
alt 2. mangaId invalid
  SV --> SV: 2.1 BadRequest
else 2. mangaId valid
  SV -> SV: 3. Validate actorId
  alt 4. actorId invalid
    SV --> SV: 4.1 BadRequest
  else 4. actorId valid
    SV -> MM: 5. findById(mangaId)
    activate MM
    MM -> DB: 6. Query manga
    activate DB
    DB --> MM: 7. Manga / null
    deactivate DB
    MM --> SV: 8. Return manga
    deactivate MM

    alt 9. Manga not found
      SV --> SV: 9.1 NotFound
    else 9. Manga found
      SV --> SV: 9.2 Continue
    end
  end
end

deactivate SV
@enduml
