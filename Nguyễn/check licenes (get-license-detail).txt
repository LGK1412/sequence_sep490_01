@startuml
skinparam {
  ActorBackgroundColor #FFFFFF
  BoundaryBackgroundColor #FFFFFF
  ControlBackgroundColor #FFFFFF
  EntityBackgroundColor #FFFFFF
  DatabaseBackgroundColor #FFFFFF
  ActorBorderColor black
  BoundaryBorderColor black
  ControlBorderColor black
  EntityBorderColor black
  DatabaseBorderColor black
}

actor "Content Moderator" as Moderator
boundary "Content Management UI" as UI
boundary Router
control MangaController
control MangaService
entity MangaModel
database MongoDB

Moderator -> UI: 1. Select a story from moderation queue
activate Moderator
activate UI

UI -> Router: 2. GET /api/manga/admin/license/{mangaId}
activate Router

Router -> MangaController: 3. getLicenseDetail(mangaId)
activate MangaController

MangaController -> MangaService: 4. getLicenseDetail(mangaId)
activate MangaService

opt Invalid mangaId
  MangaService --> MangaController: 4.1. Throw BadRequest
  MangaController --> Router: 4.1.1. HTTP 400
  Router --> UI: 4.1.2. Detail load failed
  UI --> Moderator: 4.1.3. Show error message
end

MangaService -> MangaModel: 5. findById(mangaId).select().populate().lean()
activate MangaModel
MangaModel -> MongoDB: 6. Query license detail
activate MongoDB
MongoDB --> MangaModel: 7. Story detail / null
deactivate MongoDB
MangaModel --> MangaService: 8. Return detail
deactivate MangaModel

opt Manga not found
  MangaService --> MangaController: 8.1. Throw NotFound
  MangaController --> Router: 8.1.1. HTTP 404
  Router --> UI: 8.1.2. Detail not found
  UI --> Moderator: 8.1.3. Show error message
end

MangaService --> MangaController: 9. Return normalized license detail
deactivate MangaService

MangaController --> Router: 10. HTTP 200 response
deactivate MangaController

Router --> UI: 11. Return selected story detail
deactivate Router

UI --> Moderator: 12. Display cover, files, note, review info
deactivate UI
deactivate Moderator

@enduml
