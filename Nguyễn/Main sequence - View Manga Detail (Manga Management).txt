@startuml
title View Manga Detail (Manga Management)

skinparam {
  ActorBackgroundColor #FFFFFF
  BoundaryBackgroundColor #FFFFFF
  ControlBackgroundColor #FFFFFF
  EntityBackgroundColor #FFFFFF
  DatabaseBackgroundColor #FFFFFF
  ActorBorderColor black
  BoundaryBorderColor black
  ControlBorderColor black
  EntityBorderColor black
  DatabaseBorderColor black
}

actor "Content Moderator" as MOD
boundary "Manga Management UI" as UI
boundary Router as RT
control AccessTokenGuard as ATG
control RolesGuard as RG
control MangaController as CT
control MangaService as SV
entity MangaModel as MM
database MongoDB as DB

MOD -> UI: 1. Click View
activate MOD
activate UI

UI -> RT: 2. GET /api/manga/admin/management/{mangaId}
activate RT

ref over RT, RG : 3. Check auth and role

alt 4. Unauthorized
  RT --> UI: 4.1 HTTP 401
  UI --> MOD: 4.2 Show login required
else 4. Permission denied
  RT --> UI: 4.3 HTTP 403
  UI --> MOD: 4.4 Show permission denied
else 4. Access granted
  RT -> CT: 5. getManagementDetail(mangaId)
  activate CT

  CT -> SV: 6. getManagementDetail(mangaId)
  activate SV

  alt 7. Invalid mangaId
    SV --> CT: 7.1 BadRequest
    CT --> RT: 7.2 HTTP 400
    RT --> UI: 7.3 Load failed
    UI --> MOD: 7.4 Show error
  else 7. mangaId valid
    SV -> MM: 8. findById(...).populate().lean()
    activate MM
    MM -> DB: 9. Query manga
    activate DB
    DB --> MM: 10. Manga / null
    deactivate DB
    MM --> SV: 11. Return manga
    deactivate MM

    alt 12. Manga not found
      SV --> CT: 12.1 NotFound
      CT --> RT: 12.2 HTTP 404
      RT --> UI: 12.3 Detail not found
      UI --> MOD: 12.4 Show error
    else 12. Manga found
      ref over SV : 13. Load detail metrics

      SV --> CT: 14. Manga detail response
      CT --> RT: 15. HTTP 200
      RT --> UI: 16. Return detail
      UI --> MOD: 17. Show overview, license, enforcement
    end
  end

  deactivate SV
  deactivate CT
end

deactivate RT
deactivate UI
deactivate MOD
@enduml
